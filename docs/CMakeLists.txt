# The Sphinx environment is independent of Python used by test discovery.
# Prefer an explicit override, then the active virtual environment, and only
# then CMake's general-purpose Python discovery. This avoids stale
# Python3_EXECUTABLE values cached during earlier CMake configurations.
set(ISOTHERM_SPHINX_PYTHON "" CACHE FILEPATH
    "Python executable in the environment containing Sphinx and documentation dependencies")
if(ISOTHERM_SPHINX_PYTHON)
    set(_isotherm_sphinx_python "${ISOTHERM_SPHINX_PYTHON}")
elseif(DEFINED ENV{VIRTUAL_ENV} AND NOT "$ENV{VIRTUAL_ENV}" STREQUAL "")
    set(_isotherm_sphinx_python "$ENV{VIRTUAL_ENV}/bin/python")
else()
    set(Python3_FIND_VIRTUALENV FIRST)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(_isotherm_sphinx_python "${Python3_EXECUTABLE}")
endif()
if(NOT EXISTS "${_isotherm_sphinx_python}")
    message(FATAL_ERROR
        "Documentation Python not found: ${_isotherm_sphinx_python}\n"
        "Activate a valid virtual environment or set ISOTHERM_SPHINX_PYTHON explicitly.")
endif()
file(SIZE "${_isotherm_sphinx_python}" _isotherm_python_size)
if(_isotherm_python_size EQUAL 0)
    message(FATAL_ERROR
        "Documentation Python is an empty (zero-byte) file: ${_isotherm_sphinx_python}\n"
        "Your Python virtual environment is damaged; back it up and recreate it.\n"
        "Do not try to use /usr/bin/python as a substitute for the broken environment.")
endif()
execute_process(
    COMMAND "${_isotherm_sphinx_python}" -c
            "import sphinx, pydata_sphinx_theme, bs4, matplotlib, weasyprint; print(sphinx.__version__)"
    RESULT_VARIABLE _isotherm_sphinx_result
    OUTPUT_VARIABLE _isotherm_sphinx_version
    ERROR_VARIABLE _isotherm_sphinx_error
    OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT _isotherm_sphinx_result STREQUAL "0")
    message(FATAL_ERROR
        "Sphinx/PyData/WeasyPrint/Matplotlib/BeautifulSoup are not available in ${_isotherm_sphinx_python}.\n"
        "Activate your existing WSL Python virtual environment before CMake, or use\n"
        "-DISOTHERM_SPHINX_PYTHON=/absolute/path/to/venv/bin/python.\n"
        "Original diagnostic: ${_isotherm_sphinx_error}")
endif()
message(STATUS "Sphinx ${_isotherm_sphinx_version}, Python: ${_isotherm_sphinx_python}")

set(_isotherm_docs_output "${CMAKE_CURRENT_BINARY_DIR}")
# Freshly unpacked ZIP sources can have timestamps older than cached Sphinx
# doctrees. Force a new environment so updated toctrees are never overlooked.
add_custom_target(docs-check
    COMMAND "${_isotherm_sphinx_python}" -m sphinx -E -b dummy -n -W --keep-going
            "${CMAKE_CURRENT_SOURCE_DIR}" "${_isotherm_docs_output}/check"
    COMMENT "Validate all Sphinx source documents and cross-references"
    VERBATIM)
# Recompute the toctree and rewrite HTML so ZIP timestamps cannot leave stale
# chapter navigation in the cached Sphinx build.
# Preserve Sphinx warnings in a standalone file so a failed HTML build can
# be diagnosed without relying on truncated terminal scrollback.
add_custom_target(docs-html
    COMMAND "${_isotherm_sphinx_python}" -m sphinx -E -a -b html -n -W --keep-going
            -w "${_isotherm_docs_output}/html-warnings.log"
            "${CMAKE_CURRENT_SOURCE_DIR}" "${_isotherm_docs_output}/html"
    COMMAND "${_isotherm_sphinx_python}"
            "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build_source_browser.py"
            --output "${_isotherm_docs_output}/html/source-browser"
    COMMENT "Build Isotherm++ HTML documentation and C++ source browser"
    VERBATIM)
# The PDF is composed from the same Sphinx HTML pages. WeasyPrint uses
# the approved design tokens and A4 print CSS, rather than a second LaTeX theme.
# Building the HTML first also guarantees coherent cross-references.
add_custom_target(docs-pdf
    COMMAND "${_isotherm_sphinx_python}"
            "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build_print_pdf.py"
            --html-dir "${_isotherm_docs_output}/html"
            --static-dir "${CMAKE_CURRENT_SOURCE_DIR}/_static"
            --output "${_isotherm_docs_output}/pdf/IsothermPP-Scientific-Manual.pdf"
    DEPENDS docs-html
    COMMENT "Build Isotherm++ A4 PDF from Sphinx HTML and CSS (WeasyPrint)"
    USES_TERMINAL
    VERBATIM)
