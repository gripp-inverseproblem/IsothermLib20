39/** @brief Compile-time two-parameter model contract. */
40TEST(LangmuirModel, TraitsAndConcepts) {
41 static_assert(isotherm::scalar_isotherm<Langmuir>);
42 static_assert(isotherm::differentiable_scalar_isotherm<Langmuir>);
43 static_assert(isotherm::model_traits<Langmuir>::parameter_count == 2);
44 EXPECT_EQ(isotherm::model_traits<Langmuir>::parameter_names[0], "q_max");
45 EXPECT_EQ(isotherm::model_traits<Langmuir>::parameter_names[1], "K");
46 const Langmuir model{5.0, 0.4};
47 EXPECT_DOUBLE_EQ(model.capacity(), 5.0);
48 EXPECT_DOUBLE_EQ(model.affinity(), 0.4);
49}

59/** @brief The solution grows monotonically and remains below qmax. */
60TEST(LangmuirModel, MonotonicityAndSaturation) {
61 const Langmuir model{8.0, 2.0};
62 Real previous = 0.0;
63 for (int n = 1; n <= 150; ++n) {
64 const Real concentration = static_cast<Real>(n) / 10.0;
65 const Real value = isotherm::evaluate(model, concentration);
66 EXPECT_GT(value, previous);
67 EXPECT_LT(value, 8.0);
68 previous = value;
69 }
70 EXPECT_NEAR(isotherm::evaluate(model, 1e8), 8.0, 1e-7);
71}

79/** @brief Independent exact analytic derivative and finite-difference check. */
80TEST(LangmuirModel, Derivative) {
81 const Langmuir model{10.0, 0.5};
82 EXPECT_DOUBLE_EQ(isotherm::derivative(model, 0.0), 5.0);
83 EXPECT_DOUBLE_EQ(isotherm::derivative(model, 2.0), 1.25);
84 const Real x = 1.5;
85 const Real h = 1e-5;
86 const Real numerical = (isotherm::evaluate(model, x + h) -
87 isotherm::evaluate(model, x - h)) / (2.0 * h);
88 EXPECT_NEAR(isotherm::derivative(model, x), numerical, 2e-10);
89}

91/** @brief Four linearised identities hold algebraically at C>0. */
92TEST(LangmuirModel, FourLinearisedIdentities) {
93 const Real qmax = 10.0;
94 const Real affinity = 0.5;
95 const Langmuir model{qmax, affinity};
96 for (Real concentration : {0.5, 1.0, 2.0, 5.0}) {
97 const Real q = isotherm::evaluate(model, concentration);
98 EXPECT_NEAR(concentration / q, 1.0 / (affinity * qmax) + concentration / qmax, 1e-12);
99 EXPECT_NEAR(1.0 / q, 1.0 / qmax + 1.0 / (affinity * qmax * concentration), 1e-12);
100 EXPECT_NEAR(q / concentration, affinity * qmax - affinity * q, 1e-12);
101 EXPECT_NEAR(q, qmax - (q / concentration) / affinity, 1e-12);
102 }
103}
