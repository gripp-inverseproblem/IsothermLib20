35TEST(SmithModel, MetadataAndFixedActivityTemperatureSection) {
36 static_assert(isotherm::scalar_isotherm<Smith>);
37 static_assert(isotherm::differentiable_scalar_isotherm<Smith>);
38 static_assert(!isotherm::thermal_isotherm<Smith>);
39 static_assert(isotherm::model_traits<Smith>::parameter_count == 4);
40 static_assert(!isotherm::model_traits<Smith>::temperature_dependent);
41 static_assert(isotherm::model_traits<Smith>::variable ==
42 isotherm::independent_variable::temperature);
43 static_assert(isotherm::model_traits<Smith>::parameter_names ==
44 std::array<std::string_view, 4>{"A", "B", "C", "a_w"});
45 const Smith model{1.4, -0.001, 0.55, 0.5};
46 constexpr Real temperature_k = 300.0;
47 EXPECT_NEAR(isotherm::evaluate(model, temperature_k),
48 1.1 - 0.55 * std::log(0.5), 1e-13);
49 EXPECT_NEAR(isotherm::derivative(model, temperature_k), -0.001, 1e-13);
50 EXPECT_NEAR(isotherm::evaluate(Smith{1.4, -0.001, 0.55, 0.0}, temperature_k),
51 1.1, 1e-13);
52 EXPECT_NEAR(isotherm::evaluate(model, 310.0)
53 - isotherm::evaluate(model, 300.0), -0.01, 1e-13);
54}

56TEST(SmithModel, TemperatureDerivativeMatchesCentredDifference) {
57 const Smith model{1.4, -0.001, 0.55, 0.5};
58 for (Real temperature : {273.15, 300.0, 323.15}) {
59 constexpr Real h = 0.001;
60 const Real centred = (isotherm::evaluate(model, temperature + h)
61 - isotherm::evaluate(model, temperature - h)) / (2*h);
62 EXPECT_NEAR(isotherm::derivative(model, temperature), centred, 1e-11);
63 }
64}

66TEST(SmithModel, StoredActivityChangesTheResponseAsExpected) {
67 constexpr Real h = 1e-5;
68 for (Real activity : {0.25, 0.5, 0.75}) {
69 const Smith plus{1.4, -0.001, 0.55, activity + h};
70 const Smith minus{1.4, -0.001, 0.55, activity - h};
71 const Real centred = (isotherm::evaluate(plus, 300.0)
72 - isotherm::evaluate(minus, 300.0)) / (2*h);
73 EXPECT_NEAR(centred, Real{0.55} / (Real{1} - activity), 3e-8);
74 }

77TEST(SmithModel, ScaledSensitivitiesAndIndependentReference) {
78 constexpr Real A = 0.3, B = 0.002, C = 0.1, aw = 0.4, T = 300.0;
79 const Smith model{A, B, C, aw};
80 EXPECT_NEAR(isotherm::evaluate(model, T), 0.9510825623765989, 1e-13);
81 EXPECT_NEAR(isotherm::derivative(model, T), B, 1e-13);
82 EXPECT_NEAR(A, 0.3, 1e-14);
83 EXPECT_NEAR(B * T, 0.6, 1e-14);
84 EXPECT_NEAR(-C * std::log1p(-aw), 0.051082562376599076, 1e-13);
85 EXPECT_NEAR(aw * C / (Real{1} - aw), 0.06666666666666668, 1e-13);
86}

88TEST(SmithModel, RejectsInvalidStoredParametersAndTemperatures) {
89 const Smith model{1.4, -0.001, 0.55, 0.5};
90 EXPECT_THROW((Smith{0.0, -0.001, 0.55, 0.5}), isotherm::isotherm_error);
91 EXPECT_THROW((Smith{1.4, -0.001, 0.0, 0.5}), isotherm::isotherm_error);
92 EXPECT_THROW((Smith{1.4, -0.001, 0.55, -0.1}), isotherm::isotherm_error);
93 EXPECT_THROW((Smith{1.4, -0.001, 0.55, 1.0}), isotherm::isotherm_error);
94 EXPECT_THROW((Smith{1.4, -0.001, 0.55, std::numeric_limits<Real>::infinity()}),
95 isotherm::isotherm_error);
96 EXPECT_THROW((Smith{1.4, std::numeric_limits<Real>::infinity(), 0.55, 0.5}),
97 isotherm::isotherm_error);
98 EXPECT_THROW((Smith{1.4, -0.001, 0.55, std::numeric_limits<Real>::quiet_NaN()}),
99 isotherm::isotherm_error);
100 EXPECT_THROW((void)isotherm::evaluate(model, 0.0), isotherm::isotherm_error);
101 EXPECT_THROW((void)isotherm::evaluate(model, -20.0), isotherm::isotherm_error);
102 EXPECT_THROW((void)isotherm::evaluate(model, 2000.0), isotherm::isotherm_error);
103 EXPECT_THROW((void)isotherm::evaluate(model, std::numeric_limits<Real>::infinity()),
104 isotherm::isotherm_error);
105 EXPECT_THROW((void)isotherm::derivative(model, 0.0), isotherm::isotherm_error);
106 EXPECT_THROW((void)isotherm::derivative(model, 2000.0), isotherm::isotherm_error);
107}

109TEST(SmithModel, TemperatureSpanEvaluationMatchesScalar) {
110 const Smith model{1.4, -0.001, 0.55, 0.5};
111 std::array<Real, 3> temperatures{273.15, 298.15, 323.15};
112 std::array<Real, 3> output{};
113 isotherm::evaluate_into(model, std::span<const Real>(temperatures),
114 std::span<Real>(output));
115 for (std::size_t i = 0; i < temperatures.size(); ++i) {
116 EXPECT_NEAR(output[i], isotherm::evaluate(model, temperatures[i]), 1e-13);
117 }
