30TEST(BrouersSotolongoModel, TraitsAndConcept) {
31 static_assert(isotherm::model_traits<BrouersSotolongo>::parameter_count == 3);
32 static_assert(isotherm::scalar_isotherm<BrouersSotolongo>);
33 static_assert(isotherm::differentiable_scalar_isotherm<BrouersSotolongo>);
34 static_assert(isotherm::model_traits<BrouersSotolongo>::temperature_dependent == false);
35}
36TEST(BrouersSotolongoModel, IndependentReferenceValues) {
37 const BrouersSotolongo model{5, 0.65, 1.3};
38 EXPECT_NEAR(isotherm::evaluate(model, 0.8), 1.9256031995330558, 1e-10 * (1.0 + std::abs(1.9256031995330558)));
39 EXPECT_NEAR(isotherm::evaluate(model, 2.0), 3.991009652336166, 1e-10 * (1.0 + std::abs(3.991009652336166)));
40 EXPECT_DOUBLE_EQ(isotherm::evaluate(model, 0.0), 0.0);
41}
42TEST(BrouersSotolongoModel, DerivativeMatchesCentralDifference) {
43 const BrouersSotolongo model{5, 0.65, 1.3};
44 const isotherm::Real h = 1e-5;
45 const isotherm::Real numerical = (isotherm::evaluate(model, 1.3 + h) -
46 isotherm::evaluate(model, 1.3 - h)) / (2.0 * h);
47 const isotherm::Real analytical = isotherm::derivative(model, 1.3);
48 EXPECT_NEAR(analytical, numerical, 2e-6 * (1.0 + std::abs(analytical)));
49}
50TEST(BrouersSotolongoModel, RejectsInvalidParametersAndInputs) {
51 EXPECT_THROW((BrouersSotolongo{0.0, 0.65, 1.3}), isotherm::isotherm_error);
52 EXPECT_THROW((BrouersSotolongo{5, -1.0, 1.3}), isotherm::isotherm_error);
53 const BrouersSotolongo model{5, 0.65, 1.3};
54 EXPECT_THROW(static_cast<void>(isotherm::evaluate(model, -0.1)), isotherm::isotherm_error);
55 EXPECT_THROW(static_cast<void>(isotherm::evaluate(model, std::numeric_limits<isotherm::Real>::infinity())), isotherm::isotherm_error);
56 EXPECT_THROW(static_cast<void>(isotherm::derivative(model, 0.0)), isotherm::isotherm_error);
57
58}
59TEST(BrouersSotolongoModel, SpanEvaluation) {
60 const BrouersSotolongo model{5, 0.65, 1.3};
61 std::array<isotherm::Real,3> input{0.3,0.8,2.0}, output{-1.0,-1.0,-1.0};
62 isotherm::evaluate_into(model, std::span<const isotherm::Real>(input), std::span<isotherm::Real>(output));
63 EXPECT_NEAR(output[1], 1.9256031995330558, 1e-10);
64}
