29namespace isotherm::models {
30/**
31 * @brief Four-coefficient Oswin extension:
32 * q_e = (A + B*T) * [a_w/(1-a_w)]^(1/(C + D*T)).
33 *
34 * The fourth fitted coefficient D allows the exponent to vary with T.
35 * Setting D=0 reduces the model to the three-coefficient modified Oswin
36 * equation. This extra temperature term is a library extension, not the
37 * historical Oswin (1946) formulation.
38 *
39 * T is absolute temperature (K); a_w is dimensionless water activity.
40 */
41class Oswin final {
42public:
43 Oswin(Real a, Real b, Real c, Real d)
44 : a_(a), b_(b), c_(c), d_(d) {
45 require_positive(a_, "Oswin A");
46 require_finite(b_, "Oswin B");
47 require_positive(c_, "Oswin C");
48 require_finite(d_, "Oswin D");
49 }

51 [[nodiscard]] Real evaluate(Real water_activity,
52 Real temperature_k) const {
53 const Real amplitude = amplitude_at(temperature_k);
54 const Real exponent_denom = denominator_at(temperature_k);
55 require_nonnegative(water_activity, "water activity");
56 if (water_activity >= Real{1}) {
57 throw isotherm_error(error_code::outside_domain,
58 "Water activity must be below one");
59 }
60 if (water_activity == Real{0}) return Real{0};
61 const Real ratio = water_activity / (Real{1} - water_activity);
62 return amplitude * std::pow(ratio, Real{1} / exponent_denom);
63 }
64​
65 [[nodiscard]] Real derivative(Real water_activity,
66 Real temperature_k) const {
67 require_positive(water_activity, "water activity for derivative");
68 const Real value = evaluate(water_activity, temperature_k);
69 const Real exponent_denom = denominator_at(temperature_k);
70 return value / (exponent_denom * water_activity
71 * (Real{1} - water_activity));
72 }

85 [[nodiscard]] Real amplitude_at(Real temperature_k) const {
86 require_positive(temperature_k, "absolute temperature in kelvin");
87 const Real amplitude = a_ + b_ * temperature_k;
88 require_positive(amplitude, "Oswin A + B*T");
89 return amplitude;
90 }

92 [[nodiscard]] Real denominator_at(Real temperature_k) const {
93 require_positive(temperature_k, "absolute temperature in kelvin");
94 const Real denom = c_ + d_ * temperature_k;
95 require_positive(denom, "Oswin C + D*T");
96 return denom;
