28/** @brief Legacy Holl–Krich branch; algebraically overlaps Langmuir–Freundlich. */
29class HollKrich final {
30public:
31 HollKrich(Real p1, Real p2, Real p3) : a_(p1), b_(p2), c_(p3) {
32 require_positive(a_, "HollKrich a");
33 require_positive(b_, "HollKrich b");
34 require_positive(c_, "HollKrich c");
35 }

37 [[nodiscard]] Real evaluate(Real x) const {
38 require_nonnegative(x, "equilibrium concentration");
39 const Real q = a_, k = b_, n = c_;
40 if (x == 0.0) return 0.0;
41 return q * (k * std::pow(x, n)) / (1.0 + k * std::pow(x, n));
42 }
43 [[nodiscard]] Real derivative(Real x) const {
44 require_positive(x, "concentration for the derivative");
45 const Real q = a_, k = b_, n = c_;
46 return q * k * n * std::pow(x, n - 1.0) / std::pow(1.0 + k * std::pow(x, n), 2.0);
47 }
