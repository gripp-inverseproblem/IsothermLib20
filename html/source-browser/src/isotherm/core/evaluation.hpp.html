32// The core validates finite arguments and results; individual models enforce
33// physical constraints, including positivity and any parameter restrictions.
34template<scalar_isotherm Model>
35[[nodiscard]] inline Real evaluate(const Model& model, Real x) {
36 require_finite(x, "independent variable");
37 const Real result = model.evaluate(x);
38 if (!std::isfinite(result)) {
39 throw isotherm_error(error_code::non_finite_result,
40 "Model evaluation returned a non-finite value");
41 }
42 return result;
43}
44template<thermal_isotherm Model>
45[[nodiscard]] inline Real evaluate(const Model& model, Real x, Real temperature) {
46 require_finite(x, "independent variable");
47 require_positive(temperature, "absolute temperature");
48 const Real result = model.evaluate(x, temperature);
49 if (!std::isfinite(result)) {
50 throw isotherm_error(error_code::non_finite_result,
51 "Model evaluation returned a non-finite value");
52 }
53 return result;
54}

80template<scalar_isotherm Model>
81inline void evaluate_into(const Model& model, std::span<const Real> xs,
82 std::span<Real> output) {
83 if (xs.size() != output.size()) {
84 throw isotherm_error(error_code::invalid_input,
85 "Input and output spans must have equal lengths");
86 }
87 // Per-element execution: no rollback if model evaluation fails mid-batch.
88 for (std::size_t i = 0; i < xs.size(); ++i) {
89 output[i] = evaluate(model, xs[i]);
90 }
91}
92template<thermal_isotherm Model>
93inline void evaluate_into(const Model& model, std::span<const Real> xs,
94 Real temperature, std::span<Real> output) {
95 if (xs.size() != output.size()) {
96 throw isotherm_error(error_code::invalid_input,
97 "Input and output spans must have equal lengths");
98 }
99 require_positive(temperature, "absolute temperature");
100 for (std::size_t i = 0; i < xs.size(); ++i) {
101 output[i] = evaluate(model, xs[i], temperature);
102 }
