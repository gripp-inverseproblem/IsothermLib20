18/** @brief Energy unit of R*T; temperature is always supplied in kelvin. */
19enum class gas_constant_unit {
20 joule_per_mole_kelvin, ///< R*T in J/mol (default).
21 kilojoule_per_mole_kelvin, ///< R*T in kJ/mol.
22 litre_atmosphere_per_mole_kelvin ///< R*T in L atm/mol.
23};

25/** @brief Select numerical representation of the same physical constant.
26 * Parameter energy units must match the units of R*T.
27 * The kelvin unit of temperature does not change with this selection.
28 * Reference: 2022 CODATA/NIST R = 8.314462618... J mol^-1 K^-1.
29 */
30[[nodiscard]] constexpr Real molar_gas_constant(
31 gas_constant_unit unit = gas_constant_unit::joule_per_mole_kelvin) {
32 constexpr Real R_SI = 8.31446261815324;
33 switch (unit) {
34 case gas_constant_unit::joule_per_mole_kelvin:
35 return R_SI;
36 case gas_constant_unit::kilojoule_per_mole_kelvin:
37 return R_SI / Real{1000.0};
38 case gas_constant_unit::litre_atmosphere_per_mole_kelvin:
39 // 1 L atm = 101.325 J exactly (standard atmosphere).
40 return R_SI / Real{101.325};
41 }
42 throw isotherm_error(error_code::invalid_input,
43 "Unknown molar gas constant unit");
44}
