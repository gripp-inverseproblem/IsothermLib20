39[[nodiscard]] Real numerical_scaled_sensitivity(Parameters values, Real concentration, std::size_t index) {
40 const Real nominal = values.at(index);
41 // Use a relative numerical increment, with the actual perturbed values in
42 // denominators to avoid cancellation from rounded parameter positions.
43 const Real step = std::abs(nominal) *
44 std::cbrt(std::numeric_limits<Real>::epsilon());
45 const Real upper = nominal + step;
46 const Real lower = nominal - step;
47 const Real baseline = uptake_from_class(values, concentration);
48 bool upper_valid = false;
49 bool lower_valid = false;
50 Real upper_value = Real{0};
51 Real lower_value = Real{0};
52 try {
53 values[index] = upper;
54 upper_value = uptake_from_class(values, concentration);
55 upper_valid = std::isfinite(upper_value);
56 } catch (const isotherm::isotherm_error&) { }
57 try {
58 values[index] = lower;
59 lower_value = uptake_from_class(values, concentration);
60 lower_valid = std::isfinite(lower_value);
61 } catch (const isotherm::isotherm_error&) { }
62 if (upper_valid && lower_valid)
63 return nominal * (upper_value - lower_value) / (upper - lower);
64 if (upper_valid)
65 return nominal * (upper_value - baseline) / (upper - nominal);
66 if (lower_valid)
67 return nominal * (baseline - lower_value) / (nominal - lower);
68 throw std::runtime_error("No valid parameter perturbation for numerical sensitivity");
69}
