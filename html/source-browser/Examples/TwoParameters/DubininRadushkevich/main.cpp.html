26// Parameter sensitivities come from central differences of class evaluations.
27// Temperature is held fixed; it is an independent input, not a fitted parameter.
28[[nodiscard]] isotherm::Real scaled_sensitivity_capacity(
29 isotherm::Real capacity, isotherm::Real beta,
30 isotherm::Real concentration, isotherm::Real temperature) {
31 using isotherm::Real;
32 const Real step = capacity * std::cbrt(std::numeric_limits<Real>::epsilon());
33 const isotherm::models::DubininRadushkevich lower{capacity - step, beta};
34 const isotherm::models::DubininRadushkevich upper{capacity + step, beta};
35 return capacity * (isotherm::evaluate(upper, concentration, temperature) -
36 isotherm::evaluate(lower, concentration, temperature)) /
37 ((capacity + step) - (capacity - step));
38}

40[[nodiscard]] isotherm::Real scaled_sensitivity_beta(
41 isotherm::Real capacity, isotherm::Real beta,
42 isotherm::Real concentration, isotherm::Real temperature) {
43 using isotherm::Real;
44 const Real step = beta * std::cbrt(std::numeric_limits<Real>::epsilon());
45 const isotherm::models::DubininRadushkevich lower{capacity, beta - step};
46 const isotherm::models::DubininRadushkevich upper{capacity, beta + step};
47 return beta * (isotherm::evaluate(upper, concentration, temperature) -
48 isotherm::evaluate(lower, concentration, temperature)) /
49 ((beta + step) - (beta - step));
50}

54/** @brief Generate reproducible values, slopes and numerical sensitivities. */
55int main() {
56 using isotherm::Real;
57 constexpr Real capacity = 2.7;
58 constexpr Real beta = 0.000001;
59 constexpr Real temperature = 298.15; // ALWAYS kelvin (K).
60 // Equivalent choice: R in kJ/(mol K) requires beta in mol^2/kJ^2.
61 const isotherm::models::DubininRadushkevich model_kj{
62 capacity, beta * Real{1000000.0},
63 isotherm::gas_constant_unit::kilojoule_per_mole_kelvin};
64 // Another equivalent choice is R in L atm/(mol K):
65 const isotherm::models::DubininRadushkevich model_latm{
66 capacity, beta * Real{101.325} * Real{101.325},
67 isotherm::gas_constant_unit::litre_atmosphere_per_mole_kelvin};
68 // These local models demonstrate unit selection, without altering the CSV
69 // schema consumed by the figures and reproducibility checks.
70 static_cast<void>(model_kj);
71 static_cast<void>(model_latm);
72 const isotherm::models::DubininRadushkevich model{capacity, beta};
73 std::cout << "concentration,uptake,derivative,sensitivity_qs,sensitivity_beta\n"
74 << std::setprecision(17);
75 // Preserve the existing concentration range and its 81 samples.
76 for (int index = 0; index <= 80; ++index) {
77 const Real concentration = 0.04 + static_cast<Real>(index) / 65.0;
78 std::cout << concentration << ','
79 << isotherm::evaluate(model, concentration, temperature) << ','
80 << isotherm::derivative(model, concentration, temperature) << ','
81 << scaled_sensitivity_capacity(capacity, beta, concentration, temperature) << ','
82 << scaled_sensitivity_beta(capacity, beta, concentration, temperature) << '\n';
83 }
